cmake_minimum_required(VERSION 3.24)
if(APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum supported macOS")
endif()
project(SunEarthMoon VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Distribute a standalone executable rather than requiring a GLFW/runtime install.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git
    GIT_TAG a74efa0d5628b74adc0426af4c5710e287fa7c2c)
FetchContent_Declare(glm GIT_REPOSITORY https://github.com/g-truc/glm.git
    GIT_TAG 0af55ccecd98d4e5a8d1fad7de25ba429d60e863)
FetchContent_MakeAvailable(glfw glm)
include(cmake/EmbedTextures.cmake)
embed_textures("${CMAKE_CURRENT_BINARY_DIR}/generated/texture_data.hpp")
add_executable(sun-earth-moon MACOSX_BUNDLE src/main.cpp src/renderer.cpp)
target_include_directories(sun-earth-moon PRIVATE src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(sun-earth-moon SYSTEM PRIVATE "${glfw_SOURCE_DIR}/deps" third_party/stb)
target_link_libraries(sun-earth-moon PRIVATE glfw glm::glm)
target_compile_definitions(sun-earth-moon PRIVATE GLFW_INCLUDE_NONE)
if(MSVC)
    target_compile_options(sun-earth-moon PRIVATE /W4)
else()
    target_compile_options(sun-earth-moon PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_target_properties(sun-earth-moon PROPERTIES
    MACOSX_BUNDLE_BUNDLE_NAME "Sun Earth Moon"
    MACOSX_BUNDLE_GUI_IDENTIFIER "io.github.cubelander.sun-earth-moon"
    MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
include(CTest)
if(BUILD_TESTING)
    add_executable(orbit-tests tests/orbit_tests.cpp)
    target_include_directories(orbit-tests PRIVATE src)
    target_link_libraries(orbit-tests PRIVATE glm::glm)
    add_test(NAME orbital-invariants COMMAND orbit-tests)
    add_executable(sphere-tests tests/sphere_tests.cpp)
    target_include_directories(sphere-tests PRIVATE src)
    target_link_libraries(sphere-tests PRIVATE glm::glm)
    add_test(NAME sphere-geometry COMMAND sphere-tests)
endif()
install(TARGETS sun-earth-moon BUNDLE DESTINATION . RUNTIME DESTINATION .)
install(FILES README.md LICENSE THIRD_PARTY.md DESTINATION .)
install(DIRECTORY docs/screenshots DESTINATION docs FILES_MATCHING PATTERN "*.png")
install(FILES "${glfw_SOURCE_DIR}/LICENSE.md" DESTINATION licenses RENAME GLFW.txt)
install(FILES third_party/glad/LICENSE DESTINATION licenses RENAME GLAD.txt)
install(FILES "${glm_SOURCE_DIR}/copying.txt" DESTINATION licenses RENAME GLM.txt)
install(FILES assets/textures/ATTRIBUTION.md DESTINATION licenses RENAME TEXTURES.md)
install(FILES third_party/stb/LICENSE DESTINATION licenses RENAME stb.txt)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "sun-earth-moon-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
include(CPack)
